\section{Feature Highlights}

\subsection{Application Modes}
\begin{itemize}
    \item Offline: import of recordings, processing and evaluation of hours and days of data
    \item Live: direct display of current airspace situation from network ASTERIX (UDP)
\end{itemize}

\subsection{Import}
\begin{itemize}
    \item Customizable ASTERIX import from file recordings (RAW/Netto, IOSS, RFF), PCAP and JSON to up to 4 lines
    \item (D)GPS trail import from NMEA
    \item Sector import from ESRI Shapefiles, GML, KML and GeoJSON
    \item Supported data source types: Radar, MLAT / WAM, ADS-B, System Tracker, Reference Trajectory
    \item Data Context: named surveillance context grouping data sources, sectors, FFTs, ASTERIX decoding and colors
    \item 24h+ recordings using EPOCH timestamps
\end{itemize}

\subsection{Visualization}
\begin{itemize}
    \item Multiple Views synchronized over one shared dataset, with cross-View selection and configurable View Presets
    \item Geographic View: customizable osgEarth map and OpenSceneGraph display, automatic labeling, time-filtered display, low memory footprint
    \item Table View: textual data inspection with sortable columns and CSV export
    \item Histogram View, Grid View, Scatter Plot View: distribution and correlation of any numeric variables
    \item Global Color Mode: consistent coloring by DSType, DBContent, Data Source or Data Source + Line ID
    \item Shared Layers Tree controlling per-DBContent, per-Data-Source and per-line visibility
    \item Persistent View Points for efficient revisiting of points of interest
\end{itemize}

\subsection{Analysis}
\begin{itemize}
    \item High-performance processing with low memory footprint
    \item Predefined filters and simple custom filter generation
    \item Per-DSType Analyze Data Source task, including MLAT sensor analysis with Contributing Receivers
    \item ARTAS track association (TRI) analysis
\end{itemize}

\subsection{Reference Data}
\begin{itemize}
    \item GPS Trail import from test drives or flights
    \item Reference Trajectory reconstruction from Traffic of Opportunity
    \item Basic reconstructor (free): secondary attribute association with configurable distance scoring, Linear Uniform Motion Kalman filter
    \item Advanced reconstructor (professional license): probabilistic association, radar bias estimation and correction, ADS-B-based slant range correction, IMM filter with RTS smoother
\end{itemize}

\subsection{Evaluation}
\begin{itemize}
    \item Standard compliance against EUROCAE ED-116, ED-87C/D/E, ED-117/A, ED-142 and custom standards
    \item Generalized comparison of test data versus reference data, with results inspectable at several levels of detail
    \item Automatic and manual removal of targets (short tracks, VFR, ...)
    \item Export of results as PDF, DOCX or JSON reports
\end{itemize}

\subsection{Semi-automated Processing}
\begin{itemize}
    \item Command line options for common processing tasks
    \item Runtime Command interface (TCP, JSON) for driving COMPASS from external tools
    \item Browser-based intranet access via the separate COMPASS Portal companion application
\end{itemize}
